# Darwin host tool selection is resolved before QEMU configure runs.
#
# scripts/macos-builder.bash supplies Apple ar/nm/ranlib/strip as defaults
# while preserving explicit overrides such as the coherent llvm-ar, llvm-nm,
# and llvm-ranlib installed by BOOTSTRAP_NATIVE_LLVM.  configure then records
# those exact tools in config-meson.cross.  Do not override them in this later
# native file: doing so can mix Apple archive tools with LLVM LTO objects and
# hide the toolchain selected by the public build entry.

[binaries]
# Intentionally empty.  config-meson.cross owns Darwin binary selection.

[built-in options]
# Meson's b_lto_threads=0 means automatic LTO backend parallelism, which can
# scale to the host CPU count.  Keep Darwin links bounded to one LTO worker by
# default so native Apple Silicon builds do not turn one link into a sustained
# all-core power spike.  Command-line -Db_lto_threads=N remains an explicit
# escape hatch for performance experiments.
b_lto_threads = 1
