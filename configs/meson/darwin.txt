# Darwin host tool selection is resolved before QEMU configure runs.
#
# scripts/macos-builder.bash owns archive-tool selection.  Explicit choices
# remain authoritative; a complete llvm-ar/llvm-ranlib/llvm-nm family beside
# the selected Clang is preferred; otherwise PATH-independent Apple tools are
# the fallback.  configure records that exact choice in config-meson.cross.
# Do not override it here: doing so can mix archive tools with LTO objects from
# another compiler/toolchain revision.

[binaries]
# Intentionally empty.  config-meson.cross owns Darwin binary selection.

[built-in options]
# Keep standalone/native-file use on Meson's automatic setting. The WHP
# build entry overrides this with -Db_lto_threads=$JOBS on macOS so ThinLTO
# cannot escape the compile power budget. Positive values make Clang add an
# explicit -flto-jobs=N request; zero remains the safe fallback for direct
# configure invocations which bypass WHP's power policy.
b_lto_threads = 0

# Mach-O's LLVM toolchain supports ThinLTO and an incremental object cache.
# Prefer that mode whenever QEMU enables LTO so unchanged ThinLTO backends can
# be reused across incremental links.  These options are inert when b_lto=false.
b_lto_mode = 'thin'
b_thinlto_cache = true
