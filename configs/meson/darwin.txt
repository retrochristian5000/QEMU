# Darwin host tool selection is resolved before QEMU configure runs.
#
# scripts/macos-builder.bash supplies Apple ar/nm/ranlib/strip as defaults
# while preserving explicit overrides such as the coherent llvm-ar, llvm-nm,
# and llvm-ranlib installed by BOOTSTRAP_NATIVE_LLVM.  configure then records
# those exact tools in config-meson.cross.  Do not override them in this later
# native file: doing so can mix Apple archive tools with LLVM LTO objects and
# hide the toolchain selected by the public build entry.

[binaries]
# Intentionally empty.  config-meson.cross owns Darwin binary selection.

[built-in options]
# Keep LTO backend parallelism on Meson's automatic setting.  Positive
# b_lto_threads values make Clang add an explicit -flto-jobs=N request; on
# Darwin the driver lowers that request to the linker's -threads=N backend
# option.  A value of one therefore serializes ThinLTO and leaks -threads=1
# into final link commands.  Zero leaves backend sizing automatic while
# command-line -Db_lto_threads=N remains available for explicit experiments.
b_lto_threads = 0

# Mach-O's LLVM toolchain supports ThinLTO and an incremental object cache.
# Prefer that mode whenever QEMU enables LTO so unchanged ThinLTO backends can
# be reused across incremental links.  These options are inert when b_lto=false.
b_lto_mode = 'thin'
b_thinlto_cache = true
